                        <div class="p-3 sm:p-3.5 flex-1">
                            <h3 class="text-[12.5px] sm:text-[13.5px] font-bold text-gray-900 group-hover:text-brand transition-colors leading-snug break-words">
                                ${item.title}
                            </h3>
                        </div>
                    </article>
                `;
            }).join('');
        }

        /* Render 4 bài viết tin tức trang chủ */
        function renderArticles() {
            const container = document.getElementById('articles-grid');
            if (!container) return;
            container.innerHTML = newsArticles.map(function (art, idx) {
                const imgUrl = getDriveImageUrl(art.driveId, 800);
                return `
                    <article onclick="openArticleModal(${idx})" class="cursor-pointer bg-white rounded-2xl overflow-hidden border border-gray-200/80 shadow-sm hover:shadow-md transition-all duration-300 flex flex-col group">
                        <div class="w-full aspect-[16/10] overflow-hidden bg-gray-100">
                            <img
                                src="${imgUrl}"
                                data-drive-id="${art.driveId}"
                                alt="${art.title}"
                                loading="lazy"
                                onerror="handleDriveImageError(this)"
                                class="w-full h-full object-cover object-center group-hover:scale-105 transition-transform duration-500"
                            >
                        </div>
                        <div class="p-3 sm:p-4 flex-1 flex flex-col justify-between">
                            <div>
                                <div class="text-[11px] text-gray-400 mb-1">${art.date}</div>
                                <h3 class="text-[13px] sm:text-sm font-bold text-gray-900 group-hover:text-brand transition-colors leading-snug break-words">
                                    ${art.title}
                                </h3>
                                <p class="hidden sm:block text-xs text-gray-600 mt-2 leading-relaxed">
                                    ${art.excerpt}
                                </p>
                            </div>
                            <span class="inline-flex items-center gap-1 text-xs font-semibold text-brand mt-3">
                                <span>Xem chi tiết</span>
                                <i class="fa-solid fa-angle-right text-[10px]"></i>
                            </span>
                        </div>
                    </article>
                `;
            }).join('');
        }

        /* Điều khiển 2 banner tự trượt chuyên nghiệp */
        let currentSlide = 0;
        let slideTimer = null;

        function showSlide(index) {
            const slides = document.querySelectorAll('.hero-slide');
            const dots = document.querySelectorAll('.slide-dot');
            if (!slides.length) return;
            currentSlide = (index + slides.length) % slides.length;

            slides.forEach(function (slide, idx) {
                slide.classList.toggle('active', idx === currentSlide);
            });
            dots.forEach(function (dot, idx) {
                if (idx === currentSlide) {
                    dot.classList.remove('w-2.5', 'bg-white/50');
                    dot.classList.add('w-7', 'bg-gold');
                } else {
                    dot.classList.remove('w-7', 'bg-gold');
                    dot.classList.add('w-2.5', 'bg-white/50');
                }
            });
        }

        function startSlider() {
            clearInterval(slideTimer);
            slideTimer = setInterval(function () {
                showSlide(currentSlide + 1);
            }, 5000);
        }

        /* Lightbox xem phóng to ảnh công trình */
        function openLightbox(src, caption) {
            const modal = document.getElementById('lightbox-modal');
            document.getElementById('lightbox-img').src = src;
            document.getElementById('lightbox-img').alt = caption;
            document.getElementById('lightbox-caption').textContent = caption;
            modal.classList.remove('hidden');
            document.body.style.overflow = 'hidden';
        }

        function closeLightbox() {
            const modal = document.getElementById('lightbox-modal');
            modal.classList.add('hidden');
            document.body.style.overflow = '';
        }

        /* Modal đọc chi tiết bài viết */
        function openArticleModal(index) {
            const art = newsArticles[index];
            if (!art) return;
            document.getElementById('article-modal-title').textContent = art.title;
            document.getElementById('article-modal-date').textContent = 'Đăng ngày ' + art.date + ' • Xây Dựng Việt Ngọc';
            const imgEl = document.getElementById('article-modal-img');
            imgEl.setAttribute('data-drive-id', art.driveId);
            imgEl.setAttribute('data-fallback-step', '0');
            imgEl.onerror = function () { handleDriveImageError(imgEl); };
            imgEl.src = getDriveImageUrl(art.driveId, 1000);
            imgEl.alt = art.title;

            document.getElementById('article-modal-body').innerHTML = art.content.map(function (p) {
                return '<p>' + p + '</p>';
            }).join('');

            document.getElementById('article-modal').classList.remove('hidden');
            document.body.style.overflow = 'hidden';
        }

        function closeArticleModal() {
            document.getElementById('article-modal').classList.add('hidden');
            document.body.style.overflow = '';
        }

        /* Khởi tạo toàn bộ tính năng khi tải trang */
        document.addEventListener('DOMContentLoaded', function () {
            try {
                /* Gắn fallback cho các ảnh Drive tĩnh trên trang */
                document.querySelectorAll('.drive-img').forEach(function (img) {
                    img.addEventListener('error', function () {
                        handleDriveImageError(img);
                    });
                });

                /* Render dữ liệu thực tế */
                renderConstructionGrid('all');
                renderDanangGrid();
                renderArticles();

                /* Khởi động slider 2 banner */
                showSlide(0);
                startSlider();

                document.getElementById('prev-slide').addEventListener('click', function () {
                    showSlide(currentSlide - 1);
                    startSlider();
                });
                document.getElementById('next-slide').addEventListener('click', function () {
                    showSlide(currentSlide + 1);
                    startSlider();
                });
                document.querySelectorAll('.slide-dot').forEach(function (dot) {
                    dot.addEventListener('click', function () {
                        showSlide(Number(dot.getAttribute('data-index')));
                        startSlider();
                    });
                });

                /* Hỗ trợ vuốt cảm ứng trên điện thoại cho banner */
                let touchStartX = 0;
                const sliderEl = document.getElementById('hero-slider');
                sliderEl.addEventListener('touchstart', function (e) {
                    touchStartX = e.changedTouches[0].screenX;
                }, { passive: true });
                sliderEl.addEventListener('touchend', function (e) {
                    const diff = e.changedTouches[0].screenX - touchStartX;
                    if (Math.abs(diff) > 40) {
                        showSlide(diff < 0 ? currentSlide + 1 : currentSlide - 1);
                        startSlider();
                    }
                }, { passive: true });

                /* Menu điện thoại */
                const mobileBtn = document.getElementById('mobile-menu-btn');
                const mobileMenu = document.getElementById('mobile-menu');
                mobileBtn.addEventListener('click', function () {
                    mobileMenu.classList.toggle('hidden');
                });
                document.querySelectorAll('.mobile-nav-link').forEach(function (link) {
                    link.addEventListener('click', function () {
                        mobileMenu.classList.add('hidden');
                    });
                });

                /* Xử lý gửi form tư vấn & báo giá (không dùng alert) */
                const quoteForm = document.getElementById('quote-form');
                const feedback = document.getElementById('quote-feedback');
                quoteForm.addEventListener('submit', function (e) {
                    e.preventDefault();
                    const name = document.getElementById('q-name').value.trim();
                    const phone = document.getElementById('q-phone').value.trim();
                    const type = document.getElementById('q-type').value;
                    if (!name || !phone) return;

                    feedback.textContent = 'Cảm ơn ' + name + '! Yêu cầu báo giá hạng mục "' + type + '" đã được gửi thành công. Kỹ sư Xây Dựng Việt Ngọc sẽ liên hệ tới số ' + phone + ' trong thời gian sớm nhất.';
                    feedback.classList.remove('hidden');
                    quoteForm.reset();
                });

                /* Đóng modal bằng phím Escape */
                document.addEventListener('keydown', function (e) {
                    if (e.key === 'Escape') {
                        closeLightbox();
                        closeArticleModal();
                    }
                });
            } catch (err) {
                console.error('Lỗi khởi tạo giao diện:', err);
            }
        });
    </script>
